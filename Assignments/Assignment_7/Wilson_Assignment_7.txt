# 1. Import the Assignment_7/Utah_Religions_by_County.csv data set ####
rel <- read.csv(file = 'Utah_Religions_by_County.csv')
View(rel)
str(rel)
# 2. Clean it up into “tidy” shape ####
library(janitor)
library(tidyverse)

rel1 <- rel %>%
  clean_names()
View(rel1)

rel2 <- rel1 %>%
  pivot_longer(cols = 5:17,
               names_to = 'denomination',
               values_to = 'frequency') %>%
  pivot_longer(cols = 3:4,
               names_to = 'religious',
               values_to = 'proportion')
View(rel2)

# 3. Explore the cleaned data set with a series of figures (I want to see you exploring the data set) ####
rel2 %>%
  ggplot(aes(x = frequency, 
             y = pop_2010, 
             color = denomination)) +
  geom_point() +
  facet_wrap(~county)
# Realized not a great way to view data as population is the same across the plot...
# Revert to wider dataset to make it so that denomination frequency values are not duplicated by religious/nonreligious frequency
rel3 <- rel1 %>%
  pivot_longer(cols = 5:17,
               names_to = 'denomination',
               values_to = 'frequency')
View(rel3)

rel3 %>%
  ggplot(aes(x = denomination, 
             y = frequency)) +
  geom_bar(stat = 'identity') +
  facet_wrap(~county) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))
# Of course, LDS is the MOST common/frequent denomination across every county
# Now, wondering how I can compare frequency of denomination to total population as a correlative relationship
rel3 %>%
    ggplot(aes(x = frequency, y = pop_2010)) +
  geom_point(alpha = 0.6) + 
  geom_smooth(method = 'lm', se = F) +
  scale_y_log10() +
  facet_wrap(~denomination)
#LDS is a flat line, catholic looks like it might have some correlation, everything else is too small to see well

#I learned this from ChatGPT, very useful here!
cor_results <- rel3 %>%
  group_by(denomination) %>%
  summarize(correlation = cor(pop_2010, frequency, use = 'complete.obs'),
            p_value = cor.test(pop_2010, frequency)$p.value) %>%
  arrange(desc(abs(correlation)))

print(cor_results) #Muslim has the strongest correlation of 0.759 with a p-value of 0.00000179

rel3 %>%
  filter(denomination == 'muslim') %>%
  ggplot(aes(x = frequency, y = pop_2010)) +
  geom_point(alpha = 0.6) +
  geom_smooth(method = 'lm', se = F) + 
  scale_y_log10() +
  facet_wrap(~county)
# That plot doesn't look any more correlated than any of the other denominations

rel4 <- rel3 %>%
  arrange(desc(pop_2010)) %>%
  ggplot(aes(x = pop_2010, y = frequency, color = county)) +
  geom_point() +
  geom_smooth(method = 'lm', se = F) +
  scale_y_log10() +
  facet_wrap(~denomination)
# If I make the frequency logarithmic, which isn't a good idea because it gives infinite values, 
#     then it looks like there are downward trends between several denominations as county population increases

rel3 %>%
  ggplot(aes(x = frequency, y = non_religious)) +
  geom_point() +
  geom_smooth(method = 'lm', se = F) +
  facet_wrap(~denomination)
#Looks like there are correlations betweeen frequency of lds, episcopal church, evangelical, southern baptist, each with the frequency of non-religious frequency in a county.

cor_results1 <- rel3 %>%
  group_by(denomination) %>%
  summarize(correlation = cor(non_religious, frequency, use = 'complete.obs'),
            p_value = cor.test(non_religious, frequency)$p.value) %>%
  arrange(desc(abs(correlation)))

print(cor_results1)
# correlation and p-values are strongest for lds, then episcopal, then evangelical, then southern baptis. The rest are extremely weak correlations and p-values
